cmake_minimum_required(VERSION 3.16)

project(hy3_algotrace LANGUAGES CXX)

# ---------------------------------------------------------------------------
# hy3-algotrace — Phase 1B + Phase 2B: C++17 dataset contract validator +
# offline evaluation pipeline (hy3_algotrace)
#
# CANONICAL BUILD (CMake + CTest).
#
# NOTE — planning-party authorization (2026-08-24):
#   This CMake/CTest configuration is the canonical, cross-platform build. It
#   was NOT executed in the local environment because no CMake was available
#   there. Local functional verification was performed with the existing MSVC
#   `cl.exe` instead (see build-msvc/build.bat and docs/journal/phase-01b.md /
#   phase-02b.md). CMake/CTest verification via GitHub Actions (Windows +
#   Linux) is the authoritative cross-platform check. The build-msvc/ directory
#   (produced by build-msvc/build.bat) is git-ignored and is NOT generated by
#   this CMake config.
# ---------------------------------------------------------------------------

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Third-party dependency: vendored nlohmann/json v3.12.0 single header.
# Header is included as <nlohmann/json.hpp>; the directory below is the parent
# of the `nlohmann/` folder. No FetchContent, no network, no Python.
# Treated as SYSTEM so its warnings never pollute our own code's /W4.
set(HY3_THIRD_PARTY "${CMAKE_CURRENT_SOURCE_DIR}/third_party")

# ---- compiler flags (clear if/else, no generator expressions) -------------
if(MSVC)
  set(HY3_WARN_FLAGS /W4 /EHsc /utf-8)
else()
  set(HY3_WARN_FLAGS -Wall -Wextra -Wpedantic)
endif()

# ---- core library (no main) ------------------------------------------------
add_library(hy3_algotrace_core STATIC
    src/json_loader.cpp
    src/validator.cpp
    src/sha256.cpp
    src/prompt_exporter.cpp
    src/prediction_importer.cpp
    src/reporter.cpp
    include/hy3_algotrace/diagnostic.hpp
    include/hy3_algotrace/json_loader.hpp
    include/hy3_algotrace/validator.hpp
    include/hy3_algotrace/sha256.hpp
    include/hy3_algotrace/prompt_exporter.hpp
    include/hy3_algotrace/prediction_importer.hpp
    include/hy3_algotrace/reporter.hpp)
target_include_directories(hy3_algotrace_core
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
# third_party is required by the PUBLIC header json_loader.hpp (it includes
# <nlohmann/json.hpp>), so it must be a SYSTEM PUBLIC include on core so that
# any consumer or test linking core inherits the search path automatically.
target_include_directories(hy3_algotrace_core
    SYSTEM PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party>)
target_compile_options(hy3_algotrace_core PRIVATE ${HY3_WARN_FLAGS})

# ---- CLI executable: hy3_algotrace -----------------------------------------
add_executable(hy3_algotrace src/main.cpp)
target_link_libraries(hy3_algotrace PRIVATE hy3_algotrace_core)
target_include_directories(hy3_algotrace PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(hy3_algotrace PRIVATE ${HY3_WARN_FLAGS})

# ---- Test executables ------------------------------------------------------
add_executable(hy3_algotrace_tests tests/validator_tests.cpp)
target_link_libraries(hy3_algotrace_tests PRIVATE hy3_algotrace_core)
target_include_directories(hy3_algotrace_tests PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(hy3_algotrace_tests PRIVATE ${HY3_WARN_FLAGS})

# Phase 2B-1 PromptExporter tests (independent executable, links core only).
add_executable(prompt_exporter_tests tests/prompt_exporter_tests.cpp)
target_link_libraries(prompt_exporter_tests PRIVATE hy3_algotrace_core)
target_include_directories(prompt_exporter_tests PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(prompt_exporter_tests PRIVATE ${HY3_WARN_FLAGS})

# Phase 2B PredictionImporter tests (independent executable, links core only).
add_executable(prediction_importer_tests tests/prediction_importer_tests.cpp)
target_link_libraries(prediction_importer_tests PRIVATE hy3_algotrace_core)
target_include_directories(prediction_importer_tests PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(prediction_importer_tests PRIVATE ${HY3_WARN_FLAGS})

# Phase 2B Reporter tests (independent executable, links core only).
add_executable(reporter_tests tests/reporter_tests.cpp)
target_link_libraries(reporter_tests PRIVATE hy3_algotrace_core)
target_include_directories(reporter_tests PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(reporter_tests PRIVATE ${HY3_WARN_FLAGS})

# Phase 2B end-to-end pipeline smoke (synthetic fixtures, real dataset).
add_executable(phase2b_e2e_tests tests/phase2b_e2e_tests.cpp)
target_link_libraries(phase2b_e2e_tests PRIVATE hy3_algotrace_core)
target_include_directories(phase2b_e2e_tests PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(phase2b_e2e_tests PRIVATE ${HY3_WARN_FLAGS})

# ---- Tests -----------------------------------------------------------------
enable_testing()
# Pass the canonical data path explicitly so the real-dataset case is never
# silently skipped via ../data or CWD guessing.
add_test(NAME validator_tests
         COMMAND hy3_algotrace_tests "${CMAKE_CURRENT_SOURCE_DIR}/data")
add_test(NAME prompt_exporter_tests
         COMMAND prompt_exporter_tests "${CMAKE_CURRENT_SOURCE_DIR}/data")
add_test(NAME prediction_importer_tests
         COMMAND prediction_importer_tests "${CMAKE_CURRENT_SOURCE_DIR}/data")
add_test(NAME reporter_tests
         COMMAND reporter_tests "${CMAKE_CURRENT_SOURCE_DIR}/data")
add_test(NAME phase2b_e2e_tests
         COMMAND phase2b_e2e_tests "${CMAKE_CURRENT_SOURCE_DIR}")

# Convenience: `cmake --build . --target run_cli` runs the validator on data/.
add_custom_target(run_cli
    COMMAND hy3_algotrace validate "${CMAKE_CURRENT_SOURCE_DIR}/data"
    DEPENDS hy3_algotrace
    COMMENT "Run: hy3_algotrace validate data")
